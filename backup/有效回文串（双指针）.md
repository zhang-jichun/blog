+ www.lintcode.com/problem/415/
+ 相向双指针
+ 判断合法字符的函数 .isdigit() .isalpha()
+ 考虑特殊情况，排除非法字符
+ 跳出循环的四种情况
    + 在排除了非法字符后，左右相遇了
    + 左右未相遇，且字符相同，左右各走一步，然后相遇了
    + 左右未相遇，且字符相同，左右各走一步，仍未相遇
    + 左右未相遇，且字符不同
``` python
class Solution:
    """
    @param s: A string
    @return: Whether the string is a valid palindrome
    """
    def is_valid(self, char):
        if char.isdigit() or char.isalpha():
            return True
        return False


    def is_palindrome(self, s: str) -> bool:
        # write your code here
        if len(s) <= 1:
            return True
        
        left, right = 0, len(s) - 1
        while left < right:
            # 跳过非法字符，保证left和right指向合法字符，且左右指针未相遇
            while left < right and self.is_valid(s[left]) == False:
                left += 1
            while left < right and self.is_valid(s[right]) == False:
                right -= 1
            
            # 进行一次操作后有四种情况
            # 1.左右指针相遇，说明是回文
            if left >= right:
                return True
            # 左右指针未相遇，且左右字符相同，则两指针各走一步
            elif left < right and s[left].lower() == s[right].lower():
                left += 1
                right -= 1  
                if left >= right: # 2.走一步，相遇了
                    return True
                else:
                    continue # 3.走一步，未相遇
            # 4.左右指针未相遇，但左右字符不相同，不是回文    
            elif left < right and s[left].lower() != s[right].lower():
                return False
```
