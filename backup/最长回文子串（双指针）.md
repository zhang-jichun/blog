## 最长回文字串
+ https://www.lintcode.com/problem/200/
+ 双指针：从中间向两边
+ 两种情况：从单字符起始长度奇数，从双字符起始长度偶数
+ On2时间复杂度：回文串的起始点要遍历，回文串向左右延伸需要遍历

``` python
class Solution:
    """
    @param s: input string
    @return: a string as the longest palindromic substring
    """
    def longest_palindrome(self, s: str) -> str:
        # write your code here
        if not s:
            return ""
        if len(s) == 1:
            return s
        
        res = ""
        for left in range(0, len(s)):
            temp_res = self.get_longest_palindrome_from_mid(s, left, left)
            if len(temp_res) > len(res):
                res = temp_res
        
        for left in range(0, len(s) - 1):
            temp_res = self.get_longest_palindrome_from_mid(s, left, left + 1)
            if len(temp_res) > len(res):
                res = temp_res

        return res


    def get_longest_palindrome_from_mid(self, s, left, right):
        while left >= 0 and right <= len(s) - 1 and s[left] == s[right]:
            left -= 1
            right += 1
        
        if left == right - 1:
            return ""
        else:
            return s[left + 1 : right]
```
