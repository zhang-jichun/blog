+ www.lintcode.com/problem/891/
+ 相向双指针
+ 写函数判断一段子字符是否为回文串
    + 考虑左右指针相遇的情况，也就是单字符或者空串 
    + 不需要跳过字符，该字符串就是回文串，返回true
    + 需要跳过字符，返回差异字符下标
+ 主函数 valid_palindrome
    + 第一次调用 is_palindrome 看整体是否回文
    + 如果整体不是回文，则找到差异字符
        + 左右分别跳过一个字符再调用 is_palindrome
        + 如果是子回文串，则跳过一个字符整体也是回文串
        + 如果不是子回文串，则跳过一个字符整体也不是回文串
``` python
class Solution:
    """
    @param s: a string
    @return: whether you can make s a palindrome by deleting at most one character
    """
    def is_palindrome(self, s, l, r):
        if l >= r: # 单字符或空串
            return True

        left, right = l, r
        # 左右指针相遇，或者左右字符不同才会跳出循环
        while left < right and s[left] == s[right]:
            left += 1
            right -= 1
        
        if left >= right: # 左右相遇说明是回文串
            return True
        else: # 未相遇，且左右字符不同，返回左右指针
            return left, right


    def valid_palindrome(self, s: str) -> bool:
        # Write your code here
        temp_res_1 = self.is_palindrome(s, 0, len(s) - 1)
        if temp_res_1 == True: # 直接判断字符串是否为回文串
            return True
        
        left, right = temp_res_1 # 字符串非回文，找到不同字符处的下标
        temp_res_2 = self.is_palindrome(s, left + 1, right) # 跳过左字符
        temp_res_3 = self.is_palindrome(s, left, right - 1) # 跳过右字符

        if temp_res_2 == True or temp_res_3 == True:
            return True # 跳过字符后是回文串
        else:
            return False # 跳过字符后还不是回文串
```
