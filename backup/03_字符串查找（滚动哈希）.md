+ https://www.lintcode.com/problem/594/
+ 字符串查找的O(n+m)解法
+ 滚动哈希算法，也叫Rabin-Karp
+ 31是进位的经验值，不容易产生哈希冲突
+ 在容易超限的地方注意 %BASE
``` python
class Solution:
    """
    @param source: 
    @param target: 
    @return: return the index
    """
    def str_str2(self, source: str, target: str) -> int:
        # Write your code here
        if len(target) <= 0:
            return 0
        if len(source) <= 0 or len(source) < len(target):
            return -1

        BASE = 1000000007 # 工程上常用的大质数，中间8个0
        POWER = 1
        for _ in range(len(target)):
            POWER = (POWER * 31) % BASE # 当计算哈希需要删除左侧字符时，该字符需要乘的数

        temp_hash = 0
        target_hash = 0
        for i in range(len(source)):
            if i < len(target) - 1: # 只向右计算哈希
                temp_hash = (temp_hash * 31 + (ord(source[i]) - ord('a'))) % BASE
                target_hash = (target_hash * 31 + (ord(target[i]) - ord('a'))) % BASE
                continue
            elif i == len(target) - 1: # target哈希首次计算完成，进行第一次字符串比对
                temp_hash = (temp_hash * 31 + (ord(source[i]) - ord('a'))) % BASE
                target_hash = (target_hash * 31 + (ord(target[i]) - ord('a'))) % BASE
                if temp_hash == target_hash and source[0 : i + 1] == target:
                    return 0
            else: # source的字符窗口向右滑动，右进左出
                temp_hash = (temp_hash * 31 + (ord(source[i]) - ord('a'))) % BASE # 右进
                temp_hash = (temp_hash + BASE - (ord(source[i - len(target)]) - ord('a')) * POWER % BASE) % BASE # 左出
                if temp_hash == target_hash and source[i + 1 - len(target) : i + 1] == target:
                    return i + 1 - len(target)
            
        return -1
```

+ www.lintcode.com/problem/13/
+ 字符串查找的On2解法
+ 注意非法输入与边界下标
``` python
class Solution:
    """
    @param source: 
    @param target: 
    @return: return the index
    """
    def str_str(self, source: str, target: str) -> int:
        # Write your code here
        if len(target) <= 0:
            return 0
        if len(source) <= 0 or len(source) < len(target):
            return -1

        for i in range(len(source) - len(target) + 1):
            for j in range(len(target)):
                if source[i + j] != target[j]:
                    break
                if j == len(target) - 1:
                    return i
        
        return -1
```
