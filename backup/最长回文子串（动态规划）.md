+ 状态转移矩阵：从第一个下标到第二个下标是否为回文串
+ 初始化矩阵：对角线代表单字符肯定都是回文串，以此为基础判断奇数长度的回文串
+ 注意两字符的中缝是回文，以此为基础判断偶数长度的回文串

``` python
class Solution:
    """
    @param s: input string
    @return: a string as the longest palindromic substring
    """
    def longest_palindrome(self, s: str) -> str:
        # write your code here
        if len(s) <= 1:
            return s
        
        side_length = len(s)
        # 创建矩阵
        is_palindrome = [[False] * side_length for _ in range(side_length)]
        # 对角线代表单字符回文
        for i in range(side_length):
            is_palindrome[i][i] = True

        res_left, res_right = 0, 0
        for length in range(2, side_length + 1): # 回文长度从2开始
            for left in range(0, side_length - length + 1):
                right = left + length -1
                if s[left] != s[right]: # 左右字符不同则跳过
                    continue
                if length == 2: # 如果长度是2则是回文
                    is_palindrome[left][right] = True
                    res_left, res_right = left, right
                elif is_palindrome[left + 1][right - 1] == True: # 如果长度超过2要看减去两边的字符串是不是回文
                    is_palindrome[left][right] = True
                    res_left, res_right = left, right

        return s[res_left : res_right + 1]
```
