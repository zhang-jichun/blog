+ 哈希表法
+ 遍历列表，把值作为键，把下标作为值，添加到字典中
+ 因为列表中值是唯一的，所以可以类比成哈希值

``` python
from typing import (
    List,
)

class Solution:
    """
    @param numbers: An array of Integer
    @param target: target = numbers[index1] + numbers[index2]
    @return: [index1, index2] (index1 < index2)
    """
    def two_sum(self, numbers: List[int], target: int) -> List[int]:
        # write your code here
        if len(numbers) <= 1: # 特殊情况
            return [-1, -1]

        temp_dict = dict()
        for i, num in enumerate(numbers): # 遍历下标和值
            if target - num in temp_dict: # 如果字典中存在值，使得新值与存在的值和为target
                return [temp_dict[target - num], i] # 返回下标
            else:
                temp_dict[num] = i # 不存在满足条件的值，则该值加入字典

        return [-1, -1]
```

+ 相向双指针
+ nums = [(num, i) for i, num in enumerate(numbers)] 把值和下标组成元组，再按照值排序
+ 左右指针向内缩紧找到满足条件的值，值和下标组成的元组进而找到下标
+ sorted() 结果下标再排序

``` python
from typing import (
    List,
)

class Solution:
    """
    @param numbers: An array of Integer
    @param target: target = numbers[index1] + numbers[index2]
    @return: [index1, index2] (index1 < index2)
    """
    def two_sum(self, numbers: List[int], target: int) -> List[int]:
        # write your code here
        if len(numbers) <= 1: # 特殊情况
            return [-1, -1]

        nums = [(num, i) for i, num in enumerate(numbers)] # 值与下标组成元组，再按值排序
        nums.sort()

        left, right = 0, len(nums) - 1
        while left < right:
            if nums[left][0] + nums[right][0] == target:
                return sorted([nums[left][1], nums[right][1]]) # 结果下标排序
            elif nums[left][0] + nums[right][0] > target:
                right -= 1
            elif nums[left][0] + nums[right][0] < target:
                left += 1

        return [-1, -1]
```
